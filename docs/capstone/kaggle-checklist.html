import os, sys, json, zipfile, subprocess
from pathlib import Path

WORKDIR = Path("/kaggle/working/controlsift")
if not WORKDIR.exists():
    WORKDIR = Path("/content/controlsift")
assert WORKDIR.exists(), f"missing {WORKDIR} - need an existing session tree with adapter"
os.chdir(WORKDIR)
print("cwd =", Path.cwd())

!pip -q install -U "torchao>=0.16"

adapter = Path("results/gemma_qlora/adapter")
assert adapter.exists(), f"missing adapter at {adapter.resolve()}"

cmd = [
    sys.executable,
    "scripts/run_gemma_qlora_eval.py",
    "--split",
    "validation",
    "--limit",
    "16",
]
print(">>", " ".join(cmd))
subprocess.run(cmd, check=True)
print("QLoRA smoke eval complete")

out_zip = Path("/kaggle/working/controlsift_gpu_outputs.zip")
if not out_zip.parent.exists():
    out_zip = Path("/content/controlsift_gpu_outputs.zip")
roots = [
    Path("results/gemma_zero_shot"),
    Path("results/gemma_few_shot"),
    Path("results/gemma_qlora"),
]
written = []
with zipfile.ZipFile(out_zip, "w", compression=zipfile.ZIP_DEFLATED) as zf:
    for root in roots:
        if not root.exists():
            continue
        for path in root.rglob("*"):
            if not path.is_file():
                continue
            rel = str(path).replace("\\", "/")
            if path.suffix in {".safetensors", ".bin", ".pt", ".pth"}:
                continue
            if "adapter/" in rel and path.name.startswith(("adapter_model", "model")):
                continue
            zf.write(path, arcname=rel)
            written.append(rel)
print("Wrote", out_zip, "files:", len(written))
print("Download controlsift_gpu_outputs.zip from Output")
